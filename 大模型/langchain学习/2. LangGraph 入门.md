## 基本案例

```python
from langchain_openai import ChatOpenAI  
from typing import Annotated  
from typing_extensions import TypedDict  
  
from langgraph.graph import StateGraph, START, END  
from langgraph.graph.message import add_messages  
  
API_KEY = "??"  
# 注意链接不要搞错了，最后有v1  
BASE_URL = "https://api.siliconflow.cn/v1"  
  
llm = ChatOpenAI(  
    model='Qwen/QwQ-32B',  
    api_key=API_KEY,  
    base_url=BASE_URL  
)  
  
class State(TypedDict):  
    """这里表示messages 是一个list类型，同时可能会有些处理，比如add_messages"""  
    messages: Annotated[list, add_messages]  
  
def chatbot(state: State):  
    """这里定义一个大模型节点"""  
    return {'messages':[llm.invoke(state['messages'])]}  
  
def stream_graph_updates(user_input: str):  
    # 这里传入的json对象其实就是State对象  
    for event in graph.stream(  
            {"messages": [{"role": "user", "content": user_input}]}):  
        for value in event.values():  
            print("Assistant:", value["messages"][-1].content)  
  
if __name__ == '__main__':  
    # 这里创建一个图Graph  
    graph_builder = StateGraph(State)  
    # 给图添加相关节点  
    graph_builder.add_node('chatbot', chatbot)  
    # 起始与结束节点  
    graph_builder.add_edge(START, 'chatbot')  
    graph_builder.add_edge('chatbot', END)  
  
    # 对图进行编译  
    graph = graph_builder.compile()  
    # 输出图的完成形态  
    graph.get_graph().print_ascii()  
  
    while True:  
        try:  
            user_input = input("User: ")  
            if user_input.lower() in ["quit", "exit", "q"]:  
                print("Goodbye!")  
                break  
            stream_graph_updates(user_input)  
        except:  
            user_input = "你好呀"  
            print("User: " + user_input)  
            stream_graph_updates(user_input)  
            break
```

```python
from state import State  
from langgraph.graph import END  
  
def route_tools(state: State):  
    if isinstance(state, list):  
        ai_message = state[-1]  
    elif messages := state.get("messages", []):  
        ai_message = messages[-1] # 因为最后一条才是最近的，这里messages不是一个字符串数组，而是一个对象数组  
    else:  
        raise ValueError(f"No messages found in input state to tool_edge: {state}")  
  
    if hasattr(ai_message, "tool_calls") and len(ai_message.tool_calls) > 0:  
        return "tools"  
    return END
```


```python
# BasicToolNode.py
import json  
from langchain_core.messages import ToolMessage  
  
class BasicToolNode:  
    def __init__(self, tools: list) -> None:  
        self.tools_by_name = {tool.name: tool for tool in tools}  
  
    def __call__(self, inputs: dict):  
        # 这里是取messages中最后一个元素，也就是当前问题输入  
        if messages := inputs.get("messages", []):  
            message = messages[-1]  
        else:  
            raise ValueError('No messages found in input')  
  
        outputs = []  
        for tool_call in message.tool_calls:  
            tool_result = self.tools_by_name[tool_call['name']].invoke(  
                tool_call['args']  
            )  
            outputs.append(  
                ToolMessage(  
                    content = json.dumps(tool_result),  
                    name = tool_call['name'],  
                    tool_call_id = tool_call['id']  
                )  
            )  
        return {'messages': outputs}
```

BasicToolNode 的主要功能：
- 工具管理：通过 tools_by_name 字典管理多个工具
- 工具调用处理：处理大模型返回的工具调用请求
- 结果格式化：将工具返回的结果转换为标准的消息格式

同时还需要注意这里有两种不同的赋值：

1. `Messages := inputs.Get ("messages", [])`
   - 使用海象运算符
   - Messages 的作用域扩展到整个函数
   - 这就是为什么后面可以使用 message （它是从 messages 中获取的）
2. `Message = messages[-1]`
   - 普通赋值
   - 由于是在函数内部赋值，作用域也是整个函数

```python
# graph.py
from langgraph.graph import StateGraph, START, END  
from state import State  
from chatbotNode import chatbot_with_tool_node  
from langgraph.graph.state import CompiledStateGraph  
from conditionEdges import route_tools  
from toolTavilySearch import get_tavily_tool  
from BasicToolNode import BasicToolNode  
  
def get_graph() -> CompiledStateGraph:  
    # 这里创建一个图Graph  
    graph_builder = StateGraph(State)  
    graph_builder.add_node('chatbot', chatbot_with_tool_node)  
  
    tool_node = BasicToolNode(tools=[get_tavily_tool()])  
    graph_builder.add_node("tools", tool_node)  
  
    # chatbot 是这里设置的起始节点，route_tools 是一个条件判断，最后一个是path_map 表示  
    # 如果 route_tools 返回 tools 那么就走tools节点，否则就走END节点  
    graph_builder.add_conditional_edges(  
        "chatbot",  
        route_tools,  
        {'tools': 'tools', END: END},  
    )  
    graph_builder.add_edge('tools', 'chatbot')  
    graph_builder.add_edge(START, 'chatbot')  
  
    # 对图进行编译  
    graph = graph_builder.compile()  
    graph.get_graph().print_ascii()  
    return graph  
  
  
def graph_execute(user_input: str, graph: CompiledStateGraph):  
    # 这里传入的json对象其实就是State对象  
    for event in graph.stream(  
            {"messages": [{"role": "user", "content": user_input}]}):  
        for value in event.values():  
            print("Assistant:", value["messages"][-1].content)  
  
if __name__ == '__main__':  
    graph = get_graph()  
    while True:  
        try:  
            user_input = input("User: ")  
            if user_input.lower() in ["quit", "exit", "q"]:  
                print("Goodbye!")  
                break  
            graph_execute(user_input, graph)  
        except Exception as e:  
            print(f"有报错: {e}")  
            break
```

这里其实很简单，就是我们加入了一个工具节点，大模型会绑定这个工具，在调用的时候大模型会判断是否需要调用工具，如果需要，那么就会设置一个 `tool_calls` 的工具列表，调用完之后再作为上下文提供给大模型，大模型再结合原始问题和工具的回复重新生成一个最终的结果。

相关 [API 文档](https://python.langchain.com/api_reference/core/messages/langchain_core.messages.tool.ToolMessage.html)


## 为机器人添加记忆功能

LangGraph 通过**持久化检查点机制**解决了这个问题。如果你在编译图时提供一个 `checkpointer`，并在调用图时提供一个 `thread_id`，LangGraph 会在每一步之后自动保存状态。当你再次使用相同的 `thread_id` 调用图时，图会加载其保存的状态，从而让聊天机器人能够从上次中断的地方继续对话。

**检查点机制**比简单的聊天记忆功能强大得多——它允许你在任何时候保存和恢复复杂的状态，以进行错误恢复、人工介入的工作流、时光回溯式交互等等。但在我们深入探讨之前，让我们先添加检查点机制来实现多轮对话。

```python
# graph.py
from langgraph.graph import StateGraph, START, END  
from state import State  
from chatbotNode import chatbot_with_tool_node  
from langgraph.graph.state import CompiledStateGraph  
from conditionEdges import route_tools  
from toolTavilySearch import get_tavily_tool  
from BasicToolNode import BasicToolNode  
from langgraph.checkpoint.memory import MemorySaver  
from langgraph.prebuilt import ToolNode, tools_condition  
  
def get_graph() -> CompiledStateGraph:  
    # 这里创建一个图Graph  
    graph_builder = StateGraph(State)  
    graph_builder.add_node('chatbot', chatbot_with_tool_node)  
  
    tool_node = ToolNode(tools=[get_tavily_tool()])  
    graph_builder.add_node("tools", tool_node)  
  
    # chatbot 是这里设置的起始节点，route_tools 是一个条件判断，最后一个是path_map 表示  
    # 如果 route_tools 返回 tools 那么就走tools节点，否则就走END节点  
    graph_builder.add_conditional_edges(  
        "chatbot",  
        tools_condition,  
    )  
    graph_builder.add_edge('tools', 'chatbot')  
    graph_builder.add_edge(START, 'chatbot')  
  
    # 这是存储到内存中  
    # 还是可以使用 SqliteSaver 或 PostgresSaver 并连接到你自己的数据库  
    memory = MemorySaver()  
  
    # 对图进行编译  
    graph = graph_builder.compile(checkpointer=memory)  
    graph.get_graph().print_ascii()  
    return graph  
  
  
def graph_execute(user_input: str, graph: CompiledStateGraph, config):  
    events = graph.stream(  
        {"messages": [{"role": "user", "content": user_input}]},  
        config,  
        stream_mode='values'  
    )  
    for event in events:  
        event['messages'][-1].pretty_print()  
  
if __name__ == '__main__':  
    graph = get_graph()  
  
    config = {'configurable': {'thread_id': '1'}}  
    while True:  
        try:  
            user_input = input("User: ")  
            if user_input.lower() in ["quit", "exit", "q"]:  
                print("Goodbye!")  
                break  
            graph_execute(user_input, graph, config)  
        except Exception as e:  
            print(f"有报错: {e}")  
            break
```

这里有几个变化点

```python
memory = MemorySaver()  
# 对图进行编译  
graph = graph_builder.compile(checkpointer=memory)  
```

这里使用了默认的会话存储功能，在图编译的时候传入进去。

其次，就是将相关的工具节点和条件工具替换为了框架自带的 `ToolNode, tools_condition`。另外就是设定了一次会话

```python
config = {"configurable": {"thread_id": "1"}}
```

这个用于和其他会话进行区分。

## 人工介入

智能体可能不可靠，可能需要人工输入才能成功完成任务。同样，对于某些操作，你可能希望在执行前获得人工批准，以确保一切按预期运行。

LangGraph 的[持久化](https://langgraphcn.org/concepts/persistence)层支持人工介入的工作流，允许根据用户反馈暂停和恢复执行。此功能的主要接口是[中断](https://langgraphcn.org/concepts/human_in_the_loop/#interrupt)函数。在节点内部调用 `interrupt` 会暂停执行。通过传入一个[命令](https://langgraphcn.org/concepts/human_in_the_loop/#the-command-primitive)，可以连同人类的新输入一起恢复执行。`interrupt` 在使用体验上类似于 Python 的内置函数 `input()`，[但有一些注意事项](https://langgraphcn.org/concepts/human_in_the_loop/#interrupt)。下面我们展示一个示例。

先增加一个人工介入中断的节点

```python
# humanInterrupt.py
from langgraph.types import interrupt  
from langchain_core.tools import tool  
  
  
@tool  
def human_assistance(query: str) -> str:  
    """向人类求助"""  
    human_response = interrupt({'query': query})  
    return human_response['data']
```

## 自定义状态

到目前为止，我们一直使用仅包含一个条目（消息列表）的简单状态。这个简单的状态能满足很多需求，但如果你想在不依赖消息列表的情况下定义复杂的行为，就可以向状态中添加额外的字段。在这里，我们将演示一个新场景，在该场景中，聊天机器人使用其搜索工具查找特定信息，并将这些信息转发给人工审核。让我们让聊天机器人去查询某个实体的生日。我们将在状态中添加 `name` 和 `birthday` 键：

```python
# state.py
from typing import Annotated  
from typing_extensions import TypedDict  
from langgraph.graph.message import add_messages  
  
class State(TypedDict):  
    """这里表示messages 是一个list类型，同时可能会有些处理，比如add_messages"""  
    messages: Annotated[list, add_messages]  
    name: str  
    birthday: str
```

我们将在 `human_assistance` 工具内部填充状态键。这允许人工在信息存储到状态之前对其进行审核。我们将再次使用 `Command`，这次是从工具内部发出状态更新。有关 `Command` 的用例，请[在此处](https://langgraphcn.org/concepts/low_level/#%E5%9C%A8%E5%B7%A5%E5%85%B7%E5%86%85%E9%83%A8%E4%BD%BF%E7%94%A8)阅读更多内容。

```python
# humanInterrupt.py
from typing import Annotated  
  
import langgraph.types  
from langgraph.types import Command  
from langchain_core.messages import ToolMessage  
import langchain_core.tools  
from langchain_core.tools import InjectedToolCallId  
from langgraph.types import interrupt  
from langchain_core.tools import tool  
  
  
@tool  
def human_assistance(name: str,  
                     birthday: str,  
                     tool_call_id: Annotated[str, InjectedToolCallId]) -> str:  
    """向人类求助"""  
    human_response = interrupt({  
        'question': 'Is this correct?',  
        'name': name,  
        'birthday': birthday  
    })  
    # 如果信息正确，则更新相关内容  
    # 就是这里也是人工确认，如果是正确的，那么就输入一个correct字段，值为y开头
    if human_response.get('correct', '').lower().startswith('y'):  
        verify_name = name  
        verity_birthday = birthday  
        response = 'Correct'  
    else:  
        # 否则接收人工反馈  
        verify_name = human_response.get('name', name)  
        verity_birthday = human_response.get('birthday', birthday)  
        response = f'Made a correction: {human_response}'  
    state_update = {  
        'name': verify_name,  
        'birthday': verity_birthday,  
        'messages':[ToolMessage(response, tool_call_id=tool_call_id)]  
    }  
    # 返回一个命令更新state  
    return Command(update=state_update)
```

## 时光回溯

在典型的聊天机器人工作流程中，用户会与机器人进行一次或多次交互以完成某项任务。在前面的章节中，我们了解了如何添加记忆和引入人工干预，以便对图状态进行检查点保存并控制未来的回复。

但如果你想让用户从之前的回复开始“分支”，去探索另一种结果，该怎么办呢？或者，如果你希望用户能够“倒回”助手的工作，以纠正一些错误或尝试不同的策略（这在自主软件工程师等应用中很常见），又该如何处理呢？

你可以利用 LangGraph 内置的“时光回溯”功能来实现上述两种体验以及更多功能。
